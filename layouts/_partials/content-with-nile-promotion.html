{{- $content := .Content -}} {{- $config := hugo.Data.nile_promotions -}} {{- $key := printf "%s/%s" .Section .File.TranslationBaseName -}} {{- if and $config.enabled .IsPage (in $config.sections .Section) (not (in (slice "news" "library") .Type)) (not (in ($config.excluded_pages | default slice) $key)) (ge .WordCount (int $config.minimum_words)) (ne .Params.nile_promotion false) -}} {{- /* Track HTML depth so paragraphs inside lists, quotes and figures do not count. Keep the original tokens intact and insert only between top-level blocks. */ -}} {{- $depth := 0 -}} {{- $paragraphs := 0 -}} {{- $before := "" -}} {{- $after := "" -}} {{- $insertAt := 0 -}} {{- range findRE `(?s)|<[^>]+>|[^<]+` (string $content) -}} {{- if eq $insertAt 0 -}} {{- $before = printf "%s%s" $before . -}} {{- else -}} {{- $after = printf "%s%s" $after . -}} {{- end -}} {{- if findRE `^") -}} {{- $paragraphs = add $paragraphs 1 -}} {{- if eq $paragraphs (int $config.after_paragraph) -}}{{- $insertAt = 1 -}}{{- end -}} {{- end -}} {{- else if and (findRE `^<[a-zA-Z]` .) (not (findRE `(?i)^<(area|base|br|col|embed|hr|img|input|link|meta|param|source|track|wbr)(\s|/?>)|/\s*>$` .)) -}} {{- $depth = add $depth 1 -}} {{- end -}} {{- end -}} {{- if and (ge $paragraphs (int $config.minimum_paragraphs)) (gt $insertAt 0) -}} {{- /* Distribute the general catalogue across articles with a stable hash. This gives readers variety without cookies, animation or layout shifts. */ -}} {{- $pool := $config.default_pool | default (slice $config.collection) -}} {{- $selected := index $pool (mod (hash.FNV32a $key) (len $pool)) -}} {{- /* A narrowly matched title is more useful than a general collection. */ -}} {{- $tags := apply (.Params.tags | default slice) "strings.ToLower" "." -}} {{- range $id, $offer := $config.offers -}} {{- if and $offer.tags (intersect $tags $offer.tags) -}}{{- $selected = $id -}}{{- end -}} {{- end -}} {{- with index $config.placements $key -}}{{- $selected = . -}}{{- end -}} {{- with index $config.offers $selected -}} {{- $card := partial "nile-promotion.html" (dict "page" $ "offer" .) -}} {{- $content = printf "%s%s%s" $before $card $after | safeHTML -}} {{- end -}} {{- end -}} {{- end -}} {{- return $content -}}
