{{- $content := .Content -}} {{- $config := hugo.Data.nile_promotions -}} {{- $key := printf "%s/%s" .Section .File.TranslationBaseName -}} {{- if and $config.enabled .IsPage (in $config.sections .Section) (not (in (slice "news" "library") .Type)) (not (in ($config.excluded_pages | default slice) $key)) (ge .WordCount (int $config.minimum_words)) (ne .Params.nile_promotion false) -}} {{- /* Track HTML depth so paragraphs inside lists, quotes and figures do not count. Keep the original tokens intact and insert only between top-level blocks. */ -}} {{- $depth := 0 -}} {{- $paragraphs := 0 -}} {{- $before := "" -}} {{- $after := "" -}} {{- $insertAt := 0 -}} {{- range findRE `(?s)|<[^>]+>|[^<]+` (string $content) -}} {{- if eq $insertAt 0 -}} {{- $before = printf "%s%s" $before . -}} {{- else -}} {{- $after = printf "%s%s" $after . -}} {{- end -}} {{- if findRE `^") -}} {{- $paragraphs = add $paragraphs 1 -}} {{- if eq $paragraphs (int $config.after_paragraph) -}}{{- $insertAt = 1 -}}{{- end -}} {{- end -}} {{- else if and (findRE `^<[a-zA-Z]` .) (not (findRE `(?i)^<(area|base|br|col|embed|hr|img|input|link|meta|param|source|track|wbr)(\s|/?>)|/\s*>$` .)) -}} {{- $depth = add $depth 1 -}} {{- end -}} {{- end -}} {{- if and (ge $paragraphs (int $config.minimum_paragraphs)) (gt $insertAt 0) -}} {{- /* Keep the build-time choice as the no-JavaScript fallback. Browser rotation uses the same language-independent article hash and UTC day. */ -}} {{- $pool := $config.inline_pool | default $config.default_pool | default (slice $config.collection) -}} {{- $offers := slice -}} {{- range $pool -}} {{- with index $config.offers . -}}{{- $offers = $offers | append . -}}{{- end -}} {{- end -}} {{- if gt (len $offers) 0 -}} {{- $seed := hash.FNV32a $key -}} {{- $day := div now.Unix 86400 -}} {{- $selected := mod (add $seed $day) (len $offers) -}} {{- $fallback := partial "nile-promotion.html" (dict "page" $ "offer" (index $offers $selected)) -}} {{- $templates := "" -}} {{- range $offers -}} {{- $card := partial "nile-promotion.html" (dict "page" $ "offer" .) -}} {{- $templates = printf "%s" $templates $card -}} {{- end -}} {{- $card := printf `
%s
%s
` $seed $selected $fallback $templates ("js/nile-inline-selection.js" | relURL) -}} {{- $content = printf "%s%s%s" $before $card $after | safeHTML -}} {{- end -}} {{- end -}} {{- end -}} {{- return $content -}}
