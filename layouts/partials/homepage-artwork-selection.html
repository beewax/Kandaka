{{/* All approved regional illustrated publications get 168 hours of immediate homepage priority. Fill to three cards from the unchanged weekly rotation. No future timestamps or source-only records enter this caller's pool. */}} {{- $regional := where .items "group" "sudan-nubia" -}} {{- $at := now -}} {{- with .at }}{{ $at = time.AsTime . }}{{ end -}} {{- $recent := slice -}} {{- range $regional -}} {{- $published := time.AsTime .since -}} {{- $age := sub $at.UnixMilli $published.UnixMilli -}} {{- if and (ge $age 0) (lt $age 604800000) -}} {{- $key := printf "%020d:%s" (sub 9999999999999 $published.UnixMilli) .id -}} {{- $recent = $recent | append (dict "key" $key "item" .) -}} {{- end -}} {{- end -}} {{- $selected := slice -}} {{- $ids := slice -}} {{- range sort $recent "key" -}} {{- $item := .item -}} {{- if not (in $ids $item.id) -}} {{- $selected = $selected | append $item.value -}} {{- $ids = $ids | append $item.id -}} {{- end -}} {{- end -}} {{- $args := dict "items" $regional "limit" 3 "stride" 2 -}} {{- with .at }}{{ $args = merge $args (dict "at" .) }}{{ end -}} {{- range partial "weekly-visual-selection.html" $args -}} {{- if and (lt (len $selected) 3) (not (in $ids .source)) -}} {{- $selected = $selected | append . -}} {{- $ids = $ids | append .source -}} {{- end -}} {{- end -}} {{- return $selected -}}
